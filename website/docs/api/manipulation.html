Manipulation

str.manipulation namespace

20+ methods for modifying, trimming, padding, wrapping, and transforming strings.

Quick Import:

import { str } from '@oxog/strkit';
// Use: str.manipulation.trim('  hello  ')

// Or chainable API:
import { S } from '@oxog/strkit';
S('  hello  ').trim().value;

trim / trimStart / trimEnd

str.manipulation

Remove whitespace from string ends. Optionally specify custom characters to trim.

Example 
// Basic trim
str.manipulation.trim('  hello  ');      // 'hello'
str.manipulation.trimStart('  hello');   // 'hello'
str.manipulation.trimEnd('hello  ');     // 'hello'

// Trim custom characters
str.manipulation.trim('---hello---', '-'); // 'hello'

// Chainable API
S('  hello  ').trim().value;             // 'hello'

truncate

str.manipulation

Shorten string to specified length with customizable ellipsis and position.

Example 
// Basic truncation
str.manipulation.truncate('hello world', 8);     // 'hello...'

// Custom ellipsis
str.manipulation.truncate('hello world', 8, {
  ellipsis: '...'
}); // 'hello...'

// Middle truncation
str.manipulation.truncate('hello world', 8, {
  ellipsis: '...',
  position: 'middle'
}); // 'hel...ld'

// Chainable
S('hello world').truncate(8).value;              // 'hello...'

wrap / unwrap

str.manipulation

Wrap a string with characters or remove wrapping characters.

Example 
// Symmetric wrapping
str.manipulation.wrap('hello', '"');           // '"hello"'
str.manipulation.wrap('hello', '**');          // '**hello**'

// Asymmetric wrapping
str.manipulation.wrap('hello', '[', ']');      // '[hello]'
str.manipulation.wrap('hello', '<b>', '</b>');  // '<b>hello</b>'

// Unwrap
str.manipulation.unwrap('"hello"', '"');       // 'hello'
str.manipulation.unwrap('[hello]', '[', ']'); // 'hello'

// Chainable
S('hello').wrap('[', ']').value;               // '[hello]'

between / before / after

str.manipulation

Extract substrings relative to delimiters.

Example 
// Extract between delimiters
str.manipulation.between('<tag>content</tag>', '<tag>', '</tag>'); // 'content'
str.manipulation.between('[value]', '[', ']');                    // 'value'

// Before delimiter
str.manipulation.before('hello@world.com', '@');    // 'hello'
str.manipulation.beforeLast('a@b@c', '@');          // 'a@b'

// After delimiter
str.manipulation.after('hello@world.com', '@');     // 'world.com'
str.manipulation.afterLast('a@b@c', '@');           // 'c'

// Chainable
S('hello@world').before('@').value;             // 'hello'

pad / padStart / padEnd

str.manipulation

Pad strings to reach a target length.

Example 
// Pad start (left)
str.manipulation.padStart('5', 3, '0');    // '005'
str.manipulation.padStart('42', 5, '0');   // '00042'

// Pad end (right)
str.manipulation.padEnd('hi', 5, '!');      // 'hi!!!'
str.manipulation.padEnd('1.5', 6, '0');     // '1.5000'

// Pad both sides (center)
str.manipulation.pad('hi', 6, '*');         // '**hi**'

// Chainable
S('5').padStart(3, '0').value;           // '005'

repeat / reverse

str.manipulation

Repeat or reverse strings.

Example 
// Repeat
str.manipulation.repeat('ab', 3);      // 'ababab'
str.manipulation.repeat('-', 10);      // '----------'

// Reverse
str.manipulation.reverse('hello');     // 'olleh'
str.manipulation.reverse('12345');     // '54321'

// Chainable
S('ab').repeat(3).value;              // 'ababab'
S('hello').reverse().value;          // 'olleh'

replace / replaceAll

str.manipulation

Replace occurrences of a substring.

Example 
// Replace first occurrence
str.manipulation.replace('hello hello', 'hello', 'hi'); // 'hi hello'

// Replace all occurrences
str.manipulation.replaceAll('hello hello', 'hello', 'hi'); // 'hi hi'

// With regex
str.manipulation.replaceAll('a1b2c3', /[0-9]/g, ''); // 'abc'

// Chainable
S('hello world').replace('world', 'universe').value; // 'hello universe'
