Form Validation

Validation

Validate user input with built-in validators and entropy checking.

import { str } from '@oxog/strkit';

function validateForm(data: { email: string; username: string; password: string }) {
  const errors: string[] = [];

  if (!str.validate.isEmail(data.email)) {
    errors.push('Invalid email address');
  }

  if (!str.validate.isAlphanumeric(data.username)) {
    errors.push('Username must be alphanumeric');
  }

  // Check password strength using entropy
  if (str.analysis.entropy(data.password) < 3) {
    errors.push('Password is too weak');
  }

  return { valid: errors.length === 0, errors };
}

URL Slug Generator

Sanitization

Create SEO-friendly URLs from any title using the chain API.

import { str } from '@oxog/strkit';

function createSlug(title: string): string {
  return str.sanitize.slugify(title, {
    lowercase: true,
    separator: '-',
    transliterate: true
  });
}

createSlug('Hello World! This is a Test');
// 'hello-world-this-is-a-test'

createSlug('Cafe & Restaurant - Best Food!');
// 'cafe-restaurant-best-food'

createSlug('Turkce karakterler: sch');
// 'turkce-karakterler-iso'

Content Sanitization

Security

Sanitize user-generated content for XSS prevention and clean display.

import { str } from '@oxog/strkit';

function sanitizeComment(html: string): string {
  // Remove all HTML tags
  let clean = str.sanitize.stripHtml(html);

  // Escape any remaining HTML entities
  clean = str.sanitize.escapeHtml(clean);

  // Normalize whitespace
  clean = str.sanitize.collapseWhitespace(clean);

  // Trim to max length
  clean = str.manipulation.truncate(clean, 500, '...');

  return clean;
}

sanitizeComment('<script>alert("xss")</script>Hello <b>World</b>');
// 'Hello World'

Fuzzy Search

Similarity

Find similar strings with typo tolerance for better UX.

import { str } from '@oxog/strkit';

const products = ['iPhone 15 Pro', 'Samsung Galaxy S24', 'Google Pixel 8'];

function searchProducts(query: string) {
  const result = str.similarity.bestMatch(query, products);
  return result;
}

searchProducts('iphone');
// { target: 'iPhone 15 Pro', rating: 0.46 }

searchProducts('pixl');
// { target: 'Google Pixel 8', rating: 0.36 }

// Use different algorithms
str.similarity.jaroWinkler('hello', 'hallo');  // 0.88
str.similarity.dice('night', 'nacht');        // 0.25

Internationalization (i18n)

i18n

Format strings correctly for different locales and languages.

import { str } from '@oxog/strkit';

// Turkish locale - special i/I handling
str.case.lower('ISTANBUL', { locale: 'tr' });  // 'istanbul' (dotless i)
str.case.upper('istanbul', { locale: 'tr' });  // 'ISTANBUL' (dotted I)

// Ordinals for different locales
str.format.ordinalize(1, 'en');  // '1st'
str.format.ordinalize(1, 'fr');  // '1er'
str.format.ordinalize(1, 'de');  // '1.'
str.format.ordinalize(1, 'es');  // '1o'

// Currency formatting
str.format.currency(1234.56, { currency: 'USD', locale: 'en-US' });
// '$1,234.56'

Text Diff & Patching

Diff

Compare documents and create/apply patches like Git.

import { str } from '@oxog/strkit';

const original = `Line 1
Line 2
Line 3`;

const modified = `Line 1
Line 2 - updated
Line 3
Line 4`;

// Create a unified diff patch
const patch = str.diff.createPatch('file.txt', original, modified);

// Apply patch to original
const result = str.diff.applyPatch(original, patch);
// result === modified

// Get detailed changes
const changes = str.diff.diffLines(original, modified);
// [{ added: false, removed: false, value: 'Line 1\n' }, ...]

Reading Time Calculator

Analysis

Estimate reading and speaking time for content.

import { str } from '@oxog/strkit';

function getArticleStats(content: string) {
  return {
    wordCount: str.analysis.wordCount(content),
    charCount: str.analysis.charCount(content),
    sentences: str.analysis.sentenceCount(content),
    paragraphs: str.analysis.paragraphCount(content),
    readingTime: Math.ceil(str.analysis.readingTime(content)),
    speakingTime: Math.ceil(str.analysis.speakingTime(content))

Dynamic Pluralization

Pluralization

Handle singular/plural forms with intelligent rules.

import { str } from '@oxog/strkit';

function formatCount(count: number, noun: string): string {
  return str.plural.plural(noun, count, true);
}

formatCount(0, 'item');    // '0 items'
formatCount(1, 'item');    // '1 item'
formatCount(5, 'item');    // '5 items'

formatCount(1, 'child');   // '1 child'
formatCount(3, 'child');   // '3 children'

formatCount(1, 'person');  // '1 person'
formatCount(10, 'person'); // '10 people'

// Add custom irregular
str.plural.addIrregular('cactus', 'cacti');
formatCount(2, 'cactus');  // '2 cacti'
