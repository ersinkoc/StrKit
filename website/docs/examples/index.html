Form Validation

Validate user input with built-in validators.

import { str } from '@oxog/strkit';

function validateForm(data: { email: string; username: string; password: string }) {
  const errors: string[] = [];

  if (!str.validate.isEmail(data.email)) {
    errors.push('Invalid email address');
  }

  if (!str.validate.isAlphanumeric(data.username)) {
    errors.push('Username must be alphanumeric');
  }

  if (data.password.length < 8) {
    errors.push('Password must be at least 8 characters');
  }

  // Check password strength using entropy
  if (str.analysis.entropy(data.password) < 3) {
    errors.push('Password is too weak');
  }

  return { valid: errors.length === 0, errors };
}

URL Slug Generator

Create SEO-friendly URLs from titles.

import { S } from '@oxog/strkit';

function createSlug(title: string): string {
  return S(title)
    .trim()
    .toLowerCase()
    .replaceAll(/[^\w\s-]/g, '')  // Remove special chars
    .replaceAll(/\s+/g, '-')       // Replace spaces with dashes
    .replaceAll(/-+/g, '-')        // Remove duplicate dashes
    .value();
}

createSlug('Hello World! This is a Test');
// 'hello-world-this-is-a-test'

createSlug('  Café & Restaurant - Best Food!  ');
// 'caf-restaurant-best-food'

Content Sanitization

Sanitize user-generated content for security.

import { str } from '@oxog/strkit';

function sanitizeComment(html: string): string {
  // Remove all HTML tags
  let clean = str.sanitize.stripTags(html);

  // Escape any remaining HTML entities
  clean = str.sanitize.escapeHtml(clean);

  // Normalize whitespace
  clean = str.sanitize.normalizeWhitespace(clean);

  // Trim to max length
  clean = str.manipulation.truncate(clean, 500, '...');

  return clean;
}

sanitizeComment('<script>alert("xss")</script>Hello <b>World</b>');
// 'Hello World'

Fuzzy Search

Find similar strings with typo tolerance.

import { str } from '@oxog/strkit';

const products = ['iPhone 15 Pro', 'Samsung Galaxy S24', 'Google Pixel 8', 'OnePlus 12'];

function searchProducts(query: string) {
  const results = str.similarity.findSimilar(query, products, {
    threshold: 0.3,
    algorithm: 'levenshtein'
  });

  return results.sort((a, b) => b.score - a.score);
}

searchProducts('iphone');
// [{ match: 'iPhone 15 Pro', score: 0.46 }]

searchProducts('pixl');
// [{ match: 'Google Pixel 8', score: 0.36 }]

Internationalization (i18n)

Format strings for different locales.

import { str } from '@oxog/strkit';

// Turkish locale
str.case.lower('ISTANBUL', 'tr');      // 'ıstanbul' (dotless i)
str.case.upper('istanbul', 'tr');      // 'İSTANBUL' (dotted I)
str.case.capitalize('çay', 'tr');      // 'Çay'

// German locale
str.case.upper('straße', 'de');        // 'STRASSE' (ß → SS)

// Use current locale
str.case.title('hello world', true);   // Uses browser/system locale

Text Diff & Patching

Compare and patch text documents.

import { str } from '@oxog/strkit';

const original = `Line 1
Line 2
Line 3`;

const modified = `Line 1
Line 2 - updated
Line 3
Line 4`;

// Create a patch
const patch = str.diff.createPatch('file.txt', original, modified);

// Apply patch to original
const result = str.diff.applyPatch(original, patch);
// result === modified

// Show changes
const changes = str.diff.diffLines(original, modified);
// [
//   { type: 'equal', value: 'Line 1\n' },
//   { type: 'remove', value: 'Line 2\n' },
//   { type: 'add', value: 'Line 2 - updated\n' },
//   { type: 'equal', value: 'Line 3\n' },
//   { type: 'add', value: 'Line 4' }
// ]

Reading Time Calculator

Estimate reading time for articles.

import { str } from '@oxog/strkit';

function getArticleStats(content: string) {
  return {
    wordCount: str.analysis.wordCount(content),
    charCount: str.analysis.charCount(content),
    sentences: str.analysis.sentenceCount(content),
    paragraphs: str.analysis.paragraphCount(content),
    readingTime: Math.ceil(str.analysis.readingTime(content)),
    speakingTime: Math.ceil(str.analysis.speakingTime(content))
  };
}

const article = `Lorem ipsum dolor sit amet...`; // long article

getArticleStats(article);
// {
//   wordCount: 1500,
//   charCount: 8500,
//   sentences: 75,
//   paragraphs: 12,
//   readingTime: 8,      // 8 minutes to read
//   speakingTime: 10     // 10 minutes to speak
// }

Dynamic Pluralization

Handle singular/plural forms dynamically.

import { str } from '@oxog/strkit';

function formatCount(count: number, noun: string): string {
  return str.plural.plural(noun, count, true);
}

formatCount(0, 'item');    // '0 items'
formatCount(1, 'item');    // '1 item'
formatCount(5, 'item');    // '5 items'

formatCount(1, 'child');   // '1 child'
formatCount(3, 'child');   // '3 children'

formatCount(1, 'person');  // '1 person'
formatCount(10, 'person'); // '10 people'

// Add custom irregular
str.plural.addIrregular('cactus', 'cacti');
formatCount(2, 'cactus');  // '2 cacti'
